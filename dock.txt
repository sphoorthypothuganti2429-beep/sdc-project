FROM tomcat:9-jdk17

4.Scripted pipeline

node {

    stage('Git Checkout') {
        git branch: 'main',
            url: 'https://github.com/sphoorthypothuganti2429-beep/javaproject.git'
    }

    stage('Clean') {
        bat 'mvn clean'
    }

    stage('Compile') {
        bat 'mvn compile'
    }

    stage('Test') {
        bat 'mvn test'
    }

    stage('Package') {
        bat 'mvn package'
    }

    stage('Archive') {
        archiveArtifacts artifacts: 'target/*.jar',
                         fingerprint: true
    }

    echo 'BUILD SUCCESSFUL!'
}
